✅ POST /api/v1/client/order/state-change → Cancel booking

USER:

ADDRESSES:

ROUTING SETUP REQUIRED
==============================================

Add these routes to your RouteHelper:

class RouteHelper {
  static const String home = '/home';
  static const String services = '/services';
  static const String serviceDetail = '/service-detail/:id';
  static const String cart = '/cart';
  static const String checkout = '/checkout';
  static const String payment = '/payment';
  static const String bookingConfirmation = '/booking-confirmation';
  static const String bookingTracking = '/booking-tracking/:id';
  static const String bookings = '/bookings';
  static const String account = '/account';
  static const String profile = '/profile';
  static const String settings = '/settings';
  static const String wallet = '/wallet';
  static const String favorites = '/favorites';
  static const String inbox = '/inbox';
  static const String offers = '/offers';
  static const String coupons = '/coupons';
  
  // Getters
  static String getHomeRoute() => home;
  static String getServicesRoute() => services;
  static String getServiceDetailRoute({required int serviceId}) => '$serviceDetail?id=$serviceId';
  static String getCartRoute() => cart;
  static String getCheckoutRoute() => checkout;
  static String getPaymentRoute() => payment;
  static String getBookingConfirmationRoute({required String bookingId}) => '$bookingConfirmation?id=$bookingId';
  static String getBookingTrackingRoute({required String bookingId}) => '$bookingTracking?id=$bookingId';
  static String getBookingsRoute() => bookings;
  static String getAccountRoute() => account;
  static String getProfileRoute() => profile;
  static String getSettingsRoute() => settings;
  static String getWalletRoute() => wallet;
  static String getFavoritesRoute() => favorites;
  static String getInboxRoute() => inbox;
  static String getOffersRoute() => offers;
  static String getCouponsRoute() => coupons;
}
